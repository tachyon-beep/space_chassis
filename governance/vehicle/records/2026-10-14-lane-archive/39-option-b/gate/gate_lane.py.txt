"""One lane of the vehicle's full development gate, run as the v7 supervisor's payload (standard library only).

Usage, from the gate worktree (the supervisor's `WorkingDirectory`), with the lane's interpreter:

    <venv python> gate_lane.py LANE EVIDENCE_DIR EXPECTED_TESTS [PHASE]

`PHASE` is `all` (the default), `static` (the six steps before pytest), or `pytest`. `static` and
`pytest` exist for the **staged** allocation option: two serial checkpoints per lane, which together
run exactly the steps `all` runs, on the same frozen tree, with the same expectations. Nothing is
dropped or sharded: the pytest phase is always the whole suite in one invocation.

These are the repository's gate steps, from `governance/vehicle/practice.md`'s "The gate" and
`CONTRIBUTING.md`. They run in order, each with the lane's own interpreter (`sys.executable`). Every
step runs whatever an earlier one returned, so the evidence is complete:

| step | command | expected |
|---|---|---|
| ruff | `python -m ruff check . --no-cache` | exit 0 |
| lint | `python tools/check_vehicle.py` | exit 0, and its output contains `COMPOSES, with 273 declared debt(s)` |
| strict | `python tools/check_vehicle.py --strict` | exit 2: honest debts remain; 1, 3 or a crash is a failure |
| readiness | `python tools/plant.py --readiness` | exit 0 |
| build-order | `python tools/plant.py --build-order` | exit 0 |
| measure | `python tools/measure_clock.py --samples 10` | exit 0 |
| pytest | `python -m pytest -n 2 --max-worker-restart=0 -q -o addopts= -rs -p no:cacheprovider --junitxml=EVIDENCE_DIR/junit.xml` | exit 0; JUnit `tests == EXPECTED_TESTS`, 0 failures, 0 errors, **0 skipped**; **no crashed or replaced worker** |

**No hidden retries.** xdist replaces a crashed worker by default. `--max-worker-restart=0` stops the
replacement, but it does **not** by itself force a failing exit: a worker that crashes with no test in
hand (for example after its last case) leaves no failed case, and xdist only requests shutdown, so
pytest can exit 0 with complete, clean JUnit counts. The driver therefore fails the step on any worker
crash it can see in pytest's output (xdist 3.8.0, `xdist/dsession.py`):

- `[gwN] node down: <error>`, which xdist's terminal reporter prints at **every** verbosity when a
  worker goes down with an error (`:552`). Under `-q` this is the line that appears.
- `worker gwN crashed and worker restarting disabled`, `maximum crashed workers reached: N` and
  `replacing crashed worker gwN` (`:257–264`), and `worker 'gwN' crashed while running '<id>'` (`:436`).
  xdist prints the first three only at verbosity 0 or above, so under `-q` they are absent; they are
  matched anyway, in case verbosity changes.

A crash while a test is in hand is also a failed case in JUnit, which the count check catches.

**One deliberate difference from the practice file:** pytest runs with `-n 2`, not `-n 6`. Two workers
is the program's bound; the tier-1 gate ran the same way.

**Output.**
- Each step's combined output goes to `EVIDENCE_DIR/<lane>-<step>.out`.
- `EVIDENCE_DIR/<lane>-<phase>-gate.json` records every step's argv, exit code, duration, whether its
  expectation was met and whether a worker crash was reported, plus the JUnit counts.
- The exit code is 0 only if every expectation is met, and 1 otherwise.

This driver never retries a step.
"""

from __future__ import annotations

import json
import re
import subprocess
import sys
import time
import xml.etree.ElementTree as ET
from pathlib import Path

# Any worker crash xdist reports; see the docstring for where each comes from and when it is printed.
WORKER_CRASH = re.compile(
    r"\[gw\d+\] node down:"
    r"|worker '?gw\d+'? crashed"
    r"|replacing crashed worker"
    r"|maximum crashed workers reached"
    r"|crashed worker|worker crashed"
)


def main() -> int:
    lane, evidence, expected_tests = sys.argv[1], Path(sys.argv[2]), int(sys.argv[3])
    phase = sys.argv[4] if len(sys.argv) > 4 else "all"
    if phase not in ("all", "static", "pytest"):
        print(f"unknown phase {phase!r}", file=sys.stderr)
        return 2
    py = sys.executable
    junit = evidence / "junit.xml"
    steps = [
        ("ruff", [py, "-m", "ruff", "check", ".", "--no-cache"], 0, None),
        ("lint", [py, "tools/check_vehicle.py"], 0, "COMPOSES, with 273 declared debt(s)"),
        ("strict", [py, "tools/check_vehicle.py", "--strict"], 2, None),
        ("readiness", [py, "tools/plant.py", "--readiness"], 0, None),
        ("build-order", [py, "tools/plant.py", "--build-order"], 0, None),
        ("measure", [py, "tools/measure_clock.py", "--samples", "10"], 0, None),
        ("pytest", [py, "-m", "pytest", "-n", "2", "--max-worker-restart=0", "-q", "-o", "addopts=", "-rs", "-p", "no:cacheprovider",
                    f"--junitxml={junit}"], 0, None),
    ]
    if phase == "static":
        steps = steps[:-1]
    elif phase == "pytest":
        steps = steps[-1:]
    record = {"lane": lane, "phase": phase, "interpreter": py, "python": sys.version.split()[0], "cwd": str(Path.cwd()), "steps": []}
    ok = True
    for name, argv, want, needle in steps:
        out = evidence / f"{lane}-{name}.out"
        started = time.monotonic()
        with out.open("wb") as sink:
            code = subprocess.run(argv, stdout=sink, stderr=subprocess.STDOUT).returncode
        met = code == want
        text = out.read_text(errors="replace")
        if needle is not None:
            met = met and needle in text
        crashed = name == "pytest" and WORKER_CRASH.search(text) is not None
        met = met and not crashed
        record["steps"].append({"step": name, "argv": argv, "exit": code, "expected_exit": want, "expected_text": needle,
                                "met": met, "worker_crash_reported": crashed, "seconds": round(time.monotonic() - started, 1)})
        ok &= met
    if phase == "static":
        record.update(junit=None, expected_tests=None, junit_met=None, passed=ok)
        (evidence / f"{lane}-{phase}-gate.json").write_text(json.dumps(record, indent=2) + "\n")
        print(json.dumps({"lane": lane, "phase": phase, "passed": ok}), flush=True)
        return 0 if ok else 1
    counts = None
    try:
        root = ET.parse(junit).getroot()
        suite = root.find("testsuite") if root.tag == "testsuites" else root
        counts = {k: int(suite.get(k, 0)) for k in ("tests", "failures", "errors", "skipped")}
    except (OSError, ET.ParseError) as exc:
        counts = {"error": f"{type(exc).__name__}: {exc}"}
    record["junit"] = counts
    record["expected_tests"] = expected_tests
    counts_met = counts == {"tests": expected_tests, "failures": 0, "errors": 0, "skipped": 0}
    record["junit_met"] = counts_met
    ok &= counts_met
    record["passed"] = ok
    (evidence / f"{lane}-{phase}-gate.json").write_text(json.dumps(record, indent=2) + "\n")
    print(json.dumps({"lane": lane, "phase": phase, "passed": ok, "junit": counts}), flush=True)
    return 0 if ok else 1


if __name__ == "__main__":
    sys.exit(main())
